lang.mem.soft

Atomic accesses of integers wider than an address, for a processor whose atomic instructions reach one word (lang.lower.soft): on Thumb, where Arm's A profile has exclusive loads and stores of two words (lang.arch.thumb.Feature.application), loops of them between barriers, as the word's are made, which makes each sequentially consistent whatever order it asks. No other processor has one written yet, and reaching one there is an error while compiling.

value load

load := unsafe ($T: type, p: *T, order: lang.mem.Order) -> T { $if _paired(T) { at := lang.mem.address(T, p); mut (low, high): (u32, u32) = (0, 0); unsafe lang.arch.asm({ ..lang.arch.thumb.names; dmb(sy); ldrexd(out(low).at(r2), out(high).at(r3), mem(in(at).at(r0))); dmb(sy); }); _joined(T, low, high) } else { $lang.diag.error(.{ .msg = _unsupported }) } }

value store

store := unsafe ($T: type, p: *T, value: T, order: lang.mem.Order) { $if _paired(T) { at := lang.mem.address(T, p); (low, high) := _words(T, value); unsafe lang.arch.asm({ ..lang.arch.thumb.names; push(r4, r5); dmb(sy); again := here(); ldrexd(r4, r5, mem(in(at).at(r0))); strexd(r1, in(low).at(r2), in(high).at(r3), mem(r0)); cmp(r1, 0); b(ne, again); dmb(sy); pop(r4, r5); }); } else { $lang.diag.error(.{ .msg = _unsupported }); } }

value swap

swap := unsafe ($T: type, p: *T, value: T, order: lang.mem.Order) -> T { $if _paired(T) { at := lang.mem.address(T, p); (low, high) := _words(T, value); mut (old_low, old_high): (u32, u32) = (0, 0); unsafe lang.arch.asm({ ..lang.arch.thumb.names; push(r4, r5); dmb(sy); again := here(); ldrexd(r4, r5, mem(in(at).at(r0))); strexd(r1, in(low).at(r2), in(high).at(r3), mem(r0)); cmp(r1, 0); b(ne, again); dmb(sy); mov(out(old_low).at(r0), r4); mov(out(old_high).at(r1), r5); pop(r4, r5); }); _joined(T, old_low, old_high) } else { $lang.diag.error(.{ .msg = _unsupported }) } }

value compare_swap

compare_swap := unsafe ($T: type, p: *T, expected: T, new: T, order: lang.mem.Order) -> T { $if _paired(T) { at := lang.mem.address(T, p); (low, high) := _words(T, expected); (new_low, new_high) := _words(T, new); wanted: [2]u32 = [new_low, new_high]; from := lang.mem.address([2]u32, lang.mem.pointer_to([2]u32, &wanted)); mut (old_low, old_high): (u32, u32) = (0, 0); unsafe lang.arch.asm({ ..lang.arch.thumb.names; done := label(); push(r4, r5, r6, r8); ldrd(r6, r8, mem(in(from).at(r1))); dmb(sy); again := here(); ldrexd(r4, r5, mem(in(at).at(r0))); cmp(r4, in(low).at(r2)); b(ne, done); cmp(r5, in(high).at(r3)); b(ne, done); strexd(r1, r6, r8, mem(r0)); cmp(r1, 0); b(ne, again); place(done); dmb(sy); mov(out(old_low).at(r0), r4); mov(out(old_high).at(r1), r5); pop(r4, r5, r6, r8); }); _joined(T, old_low, old_high) } else { $lang.diag.error(.{ .msg = _unsupported }) } }

Stores new when p holds expected, giving what it held.
