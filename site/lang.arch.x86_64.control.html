value syscall

syscall := _x.Instruction{ .forms = &[ .{ .operands = &[], .sizes = _all, .opcode = &[0x0f, 0x05], .layout = .bare, .w = .never, .reads = &[.rax], .writes = &[.rax, .rcx, .r11], }, ] }
